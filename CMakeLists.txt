cmake_minimum_required(VERSION 3.16)
project(TIB_SLAM)

IF(NOT CMAKE_BUILD_TYPE)
  SET(CMAKE_BUILD_TYPE Release)
ENDIF()

MESSAGE("Build type: " ${CMAKE_BUILD_TYPE})

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS}  -Wall   -O3")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall   -O3")
# Do not use -march=native: TIB-SLAM links prebuilt/vendor g2o.
# Different AVX alignment choices across those binaries can produce Eigen
# misaligned accesses and seemingly random optimizer crashes.

# C++17 works with Ubuntu 24.04's GCC 13. A single project-wide standard also
# prevents ABI surprises.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_definitions(-DCOMPILEDWITHC11)

LIST(APPEND CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/cmake_modules)

find_package(OpenCV 4.4 REQUIRED COMPONENTS
  core imgproc imgcodecs highgui features2d calib3d videoio line_descriptor dnn)

MESSAGE("OPENCV VERSION:")

find_package(Eigen3 3.1.0 REQUIRED)
find_package(Pangolin REQUIRED)
find_package(realsense2)

option(BUILD_STRUCTURAL_TESTS "Build finite-difference tests for structural g2o edges" OFF)
if(BUILD_STRUCTURAL_TESTS)
  add_executable(structural_jacobian_test
    tests/StructuralJacobianTest.cc
    src/StructuralG2oTypes.cc)
  target_link_libraries(structural_jacobian_test
    ${OpenCV_LIBS}
    ${PROJECT_SOURCE_DIR}/Thirdparty/g2o/lib/libg2o.so)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(structural_jacobian_test PRIVATE
      -fsanitize=address,undefined -fno-omit-frame-pointer)
    set_target_properties(structural_jacobian_test PROPERTIES
      LINK_FLAGS "-fsanitize=address,undefined")
  endif()

  add_executable(gaussian_policy_test
    tests/GaussianPolicyTest.cc
    src/ObservabilityGaussianMapper.cc)
  target_link_libraries(gaussian_policy_test
    ${OpenCV_LIBS}
    pthread)

  add_executable(direct_pose_refiner_test
    tests/DirectRgbdPoseRefinerTest.cc
    src/DirectRgbdPoseRefiner.cc)
  target_link_libraries(direct_pose_refiner_test
    ${OpenCV_LIBS}
    pthread)

  add_executable(information_budget_scheduler_test
    tests/InformationBudgetSchedulerTest.cc
    src/InformationBudgetScheduler.cc)
endif()
